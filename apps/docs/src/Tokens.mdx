export const flattenTokens = (node, path = []) => {
	if (!node || typeof node !== 'object') return [];

    if ('$value' in node) {
    	return [
    		{
    			name: path.join('.'),
    			value: String(node.$value),
    			type: node.$type ? String(node.$type) : '-',
    		},
    	];
    }

    return Object.entries(node).flatMap(([key, value]) => flattenTokens(value, [...path, key]));

};

export const resolveReference = (value, lookup, depth = 0) => {
	if (typeof value !== 'string') return value;
	if (depth > 8) return value;
	if (!value.startsWith('{') || !value.endsWith('}')) return value;

    const key = value.slice(1, -1);
    const next = lookup[key];

    if (!next) return value;
    return resolveReference(next, lookup, depth + 1);

};

export const copyToClipboard = async (text) => {
  if (typeof navigator !== 'undefined' && navigator.clipboard?.writeText) {
    await navigator.clipboard.writeText(text);
    return;
  }

if (typeof document === 'undefined') {
return;
}

const textarea = document.createElement('textarea');
textarea.value = text;
textarea.setAttribute('readonly', '');
textarea.style.position = 'absolute';
textarea.style.left = '-9999px';
document.body.appendChild(textarea);
textarea.select();

try {
document.execCommand('copy');
} finally {
document.body.removeChild(textarea);
}
};

# Design Tokens

Tokens live in `packages/tokens/src/tokens` and are compiled by Style Dictionary.

The React package imports generated CSS variables from `@pitchfork-ui/tokens/css`.

## CSS Custom Properties (Generated)

All design tokens are compiled to CSS custom properties in `@pitchfork-ui/tokens/css`.

### Color Custom Properties

Color variables are shown as swatches in a grid so you can quickly scan hue and contrast.

<CssColorVariableGroups groups={cssColorVariableGroups} />

### Dimension Custom Properties

Dimension variables include spacing and radius values used for layout rhythm and shape.

<h4>Spacing</h4>

<CssSpacingVariablePreview rows={cssSpacingVariableRows} />

<h4>Radius</h4>

<CssRadiusVariablePreview rows={cssRadiusVariableRows} />

### Font Custom Properties

Font variables define family, sizing, and weight choices used across components.

<CssFontVariablePreview rows={cssFontVariableRows} />

### Shadow Custom Properties

Shadow variables provide elevation depth and surface layering.

<CssShadowVariablePreview rows={cssShadowVariableRows} />

## Dark Theme Layer

The React theme adds a dark override layer using `[data-theme='dark']`, remapping semantic tokens while preserving the same variable API.

```css
[data-theme='dark'] {
  --color-semantic-background-default: var(--color-gray-900);
  --color-semantic-text-default: var(--color-gray-50);
  --color-semantic-action-primary: var(--color-brand-500);
}
```

To activate dark mode on any subtree:

```html
<div data-theme="dark">...</div>
```

## Component Alias Variables

Component-level aliases are available in the React theme layer (`packages/react/src/styles/theme.css`) to make per-component theming simpler without changing semantic token names.

Click any alias variable to copy it, then edit its value inline to preview real component changes instantly.
