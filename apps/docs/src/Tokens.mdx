export const flattenTokens = (node, path = []) => {
	if (!node || typeof node !== 'object') return [];

    if ('$value' in node) {
    	return [
    		{
    			name: path.join('.'),
    			value: String(node.$value),
    			type: node.$type ? String(node.$type) : '-',
    		},
    	];
    }

    return Object.entries(node).flatMap(([key, value]) => flattenTokens(value, [...path, key]));

};

export const typographyRows = flattenTokens(typographyTokens);
export const shadowRows = flattenTokens(shadowTokens);

export const colorValueByName = Object.fromEntries(
  colorRows.map((row) => [row.name, row.value]),
);

export const resolveReference = (value, lookup, depth = 0) => {
	if (typeof value !== 'string') return value;
	if (depth > 8) return value;
	if (!value.startsWith('{') || !value.endsWith('}')) return value;

    const key = value.slice(1, -1);
    const next = lookup[key];

    if (!next) return value;
    return resolveReference(next, lookup, depth + 1);

};

export const resolvedColorRows = colorRows.map((row) => ({
  ...row,
  resolvedValue: resolveReference(row.value, colorValueByName),
}));

export const toCssVarName = (name) =>
  `--${name
    .replace(/([a-z0-9])([A-Z])/g, '$1-$2')
    .replace(/\./g, '-')
    .toLowerCase()}`;

# Design Tokens

Tokens live in `packages/tokens/src/tokens` and are compiled by Style Dictionary.

The React package imports generated CSS variables from `@pitchfork-ui/tokens/css`.

## CSS Custom Properties (Generated)

All design tokens are compiled to CSS custom properties in `@pitchfork-ui/tokens/css`.

<CssVariableTable rows={cssVariableRows} />

## Dark Theme Layer

The React theme adds a dark override layer using `[data-theme='dark']`, remapping semantic tokens while preserving the same variable API.

```css
[data-theme='dark'] {
  --color-semantic-background-default: var(--color-gray-900);
  --color-semantic-text-default: var(--color-gray-50);
  --color-semantic-action-primary: var(--color-brand-500);
}
```

To activate dark mode on any subtree:

```html
<div data-theme="dark">...</div>
```

## Component Alias Variables

Component-level aliases are available in the React theme layer (`packages/react/src/styles/theme.css`) to make per-component theming simpler without changing semantic token names.
